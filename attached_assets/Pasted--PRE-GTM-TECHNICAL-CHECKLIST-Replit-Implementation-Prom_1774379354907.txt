```javascript
export async function middleware(request) {
  const protectedPaths = ['/dashboard', '/matches', '/introductions', '/chat', '/profile', '/settings', '/deal-room'];
  
  const isProtected = protectedPaths.some(path => request.nextUrl.pathname.startsWith(path));

}
```

### 2.3 Rate Limiting


```javascript
// lib/rateLimit.ts
const rateLimitMap = new Map();

export function rateLimit(key, maxAttempts = 5, windowMs = 60000) {
  const now = Date.now();
  const record = rateLimitMap.get(key) || { count: 0, resetAt: now + windowMs };
  
  if (now > record.resetAt) {
    record.count = 0;
    record.resetAt = now + windowMs;
  }
  
  record.count++;
  rateLimitMap.set(key, record);
  
  if (record.count > maxAttempts) {
    return { limited: true, retryAfter: Math.ceil((record.resetAt - now) / 1000) };
  }
  return { limited: false };
}
```

Apply to login and register routes:
```javascript
const ip = req.headers.get('x-forwarded-for') || 'unknown';
const limit = rateLimit(`auth:${ip}`, 5, 60000); // 5 attempts per minute
if (limit.limited) {
  return NextResponse.json(
    { error: `Too many attempts. Try again in ${limit.retryAfter} seconds.` },
    { status: 429 }
  );
}
```
